CREATE PROCEDURE UpdatePipelineRunHistory
@PipelineName NVARCHAR(100),
@LastRunTime DATETIME
AS
BEGIN

   IF EXISTS (SELECT 1 FROM PipelineRunHistory WHERE PipelineName = @PipelineName)
   BEGIN

      UPDATE PipelineRunHistory
      SET LastRunTime = @LastRunTime
      WHERE PipelineName = @PipelineName;
   END
   ELSE
   BEGIN

      INSERT INTO PipelineRunHistory (PipelineName, LastRunTime)
      VALUES (@PipelineName, @LastRunTime);
   END
END;

    PipelineName NVARCHAR(100) PRIMARY KEY,
    LastRunTime DATETIME
